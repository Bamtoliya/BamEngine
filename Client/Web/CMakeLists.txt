cmake_minimum_required(VERSION 3.21)
project(BamWebGPUClient LANGUAGES NONE)

get_filename_component(BAM_WEB_PROJECT_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../.." ABSOLUTE)
file(READ "${BAM_WEB_PROJECT_ROOT}/Config/Project.json" project_settings)
string(JSON asset_directory GET "${project_settings}" Settings AssetDirectory)
string(JSON default_startup_scene GET "${project_settings}" Settings StartupScenePath)
set(BAM_WEB_STARTUP_SCENE "" CACHE STRING "Startup scene relative to the project's asset directory")
set(startup_scene "${BAM_WEB_STARTUP_SCENE}")
if(startup_scene STREQUAL "")
    set(startup_scene "${default_startup_scene}")
endif()

file(REAL_PATH "${asset_directory}" asset_root BASE_DIRECTORY "${BAM_WEB_PROJECT_ROOT}")
file(REAL_PATH "${startup_scene}" scene_file BASE_DIRECTORY "${asset_root}")
cmake_path(IS_PREFIX asset_root "${scene_file}" NORMALIZE scene_in_assets)
if(NOT scene_in_assets OR NOT scene_file MATCHES "\\.bamscene\\.json$" OR NOT EXISTS "${scene_file}" OR IS_DIRECTORY "${scene_file}")
    message(FATAL_ERROR "Choose a saved .bamscene.json inside ${asset_root}: ${startup_scene}")
endif()

set(web_output "${CMAKE_CURRENT_BINARY_DIR}/site")
foreach(web_file IN ITEMS index.html main.js scene.js scene.wgsl)
    configure_file("${CMAKE_CURRENT_SOURCE_DIR}/${web_file}" "${web_output}/${web_file}" COPYONLY)
endforeach()
configure_file("${scene_file}" "${web_output}/StartupScene.bamscene.json" COPYONLY)
configure_file("${BAM_WEB_PROJECT_ROOT}/Config/InputActions.json" "${web_output}/InputActions.json" COPYONLY)
set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${BAM_WEB_PROJECT_ROOT}/Config/Project.json")
message(STATUS "WebGPU Client: ${startup_scene} -> ${web_output}")
